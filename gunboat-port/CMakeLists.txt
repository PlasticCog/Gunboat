cmake_minimum_required(VERSION 3.24)
project(gunboat_port VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/deps/SDL3-3.4.16/cmake")
  list(PREPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/deps/SDL3-3.4.16/cmake")
endif()
find_package(SDL3 REQUIRED CONFIG)
add_library(td3_host STATIC vendor/td3/host.c vendor/td3/platform/vga.c vendor/nuked-opl3/opl3.c)
target_include_directories(td3_host PUBLIC vendor/td3 PRIVATE vendor/nuked-opl3)
target_link_libraries(td3_host PUBLIC SDL3::SDL3)
add_executable(gunboat src/main.cpp src/assets.cpp src/simulation.cpp src/game.cpp src/original_renderer.cpp src/original_audio.cpp src/original_controls.cpp src/original_physics.cpp)
target_include_directories(gunboat PRIVATE src)
target_link_libraries(gunboat PRIVATE td3_host)
if(MSVC)
  target_compile_options(gunboat PRIVATE /W4)
else()
  target_compile_options(gunboat PRIVATE -Wall -Wextra -Wpedantic)
  if(WIN32)
    target_link_options(gunboat PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
if(WIN32)
  add_custom_command(TARGET gunboat POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:gunboat>)
endif()
enable_testing()
add_test(NAME original_assets COMMAND gunboat --game-dir "${CMAKE_CURRENT_SOURCE_DIR}/../Original DOS version" --check)
add_test(NAME native_simulation COMMAND gunboat --game-dir "${CMAKE_CURRENT_SOURCE_DIR}/../Original DOS version" --self-test)
